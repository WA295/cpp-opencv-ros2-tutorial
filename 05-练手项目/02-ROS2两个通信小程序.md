## 1. 准备：建工作空间和包

```bash
# 1. 建工作空间
mkdir -p ~/practice_ws/src
cd ~/practice_ws

# 2. 建包（一次建好，两个项目都放里面）
cd src
ros2 pkg create --build-type ament_cmake my_nodes \
    --dependencies rclcpp std_msgs sensor_msgs
```

生成的结构：

```
practice_ws/src/my_nodes/
├── CMakeLists.txt
├── package.xml
└── src/          ← 我们在这写 .cpp
```

**两个项目最后都要加进 `CMakeLists.txt` 的 `add_executable` + `install`**（见下文）。

## 2. 项目 A：talker + listener 聊天

### 目标

- `talker`：每秒发布一条 `std_msgs/String` 到话题 `/chat`。
- `listener`：订阅 `/chat`，收到就打印。

### 学到什么

- 发布者：`create_publisher` + 定时器 + `publish`
- 订阅者：`create_subscription` + 回调
- `ros2 topic` 验证闭环

### 2.1 talker.cpp

`my_nodes/src/talker.cpp`：

```cpp
#include <rclcpp/rclcpp.hpp>
#include <std_msgs/msg/string.hpp>
#include <chrono>

using namespace std::chrono_literals;

class Talker : public rclcpp::Node
{
public:
    Talker() : Node("talker"), count_(0)
    {
        pub_ = this->create_publisher<std_msgs::msg::String>("/chat", 10);

        // 每秒触发一次
        timer_ = this->create_wall_timer(1s, [this]() {
            std_msgs::msg::String msg;
            msg.data = "你好！第 " + std::to_string(++count_) + " 条消息";
            pub_->publish(msg);
            RCLCPP_INFO(this->get_logger(), "已发布: '%s'", msg.data.c_str());
        });
    }

private:
    rclcpp::Publisher<std_msgs::msg::String>::SharedPtr pub_;
    rclcpp::TimerBase::SharedPtr timer_;
    int count_;
};

int main(int argc, char** argv)
{
    rclcpp::init(argc, argv);
    rclcpp::spin(std::make_shared<Talker>());
    rclcpp::shutdown();
    return 0;
}
```

### 2.2 listener.cpp

`my_nodes/src/listener.cpp`：

```cpp
#include <rclcpp/rclcpp.hpp>
#include <std_msgs/msg/string.hpp>

class Listener : public rclcpp::Node
{
public:
    Listener() : Node("listener")
    {
        sub_ = this->create_subscription<std_msgs::msg::String>(
            "/chat", 10,
            [this](const std_msgs::msg::String::SharedPtr msg) {
                RCLCPP_INFO(this->get_logger(), "收到: '%s'", msg->data.c_str());
            });
    }

private:
    rclcpp::Subscription<std_msgs::msg::String>::SharedPtr sub_;
};

int main(int argc, char** argv)
{
    rclcpp::init(argc, argv);
    rclcpp::spin(std::make_shared<Listener>());
    rclcpp::shutdown();
    return 0;
}
```

### 2.3 加进 CMakeLists.txt

在 `add_executable(campus...)` 的位置（原文件里没有就加在 `ament_target_dependencies` 之前）：

```cmake
add_executable(talker src/talker.cpp)
ament_target_dependencies(talker rclcpp std_msgs)

add_executable(listener src/listener.cpp)
ament_target_dependencies(listener rclcpp std_msgs)

install(TARGETS talker listener
  DESTINATION lib/${PROJECT_NAME})
```

### 2.4 编译运行

```bash
cd ~/practice_ws
source /opt/ros/jazzy/setup.bash
colcon build --packages-select my_nodes
source install/setup.bash
```

**开两个终端：**

```bash
# 终端 1
ros2 run my_nodes talker

# 终端 2
ros2 run my_nodes listener
```

终端 2 会每秒打印一条收到的消息 —— **你的第一个 ROS 2 通信闭环完成！**

## 3. 项目 B：图像亮度检测节点

### 目标

- 订阅 `/camera/image_raw/compressed`（压缩图像）。
- 解码 → 转灰度 → 算平均亮度。
- 发布亮度到 `/image_brightness`（`std_msgs/Float32`）。
- （可选）显示图像。

### 学到什么

- ROS 2 + OpenCV 结合（`imdecode`）
- 订阅方同时是发布方（**中转处理节点**，最常用的节点形态）

### 3.1 brightness_node.cpp

`my_nodes/src/brightness_node.cpp`：

```cpp
#include <rclcpp/rclcpp.hpp>
#include <sensor_msgs/msg/compressed_image.hpp>
#include <std_msgs/msg/float32.hpp>
#include <opencv2/opencv.hpp>

class BrightnessNode : public rclcpp::Node
{
public:
    BrightnessNode() : Node("brightness_node")
    {
        sub_ = this->create_subscription<sensor_msgs::msg::CompressedImage>(
            "/camera/image_raw/compressed", 1,
            [this](const sensor_msgs::msg::CompressedImage::SharedPtr msg) {
                this->onImage(msg);
            });

        pub_ = this->create_publisher<std_msgs::msg::Float32>(
            "/image_brightness", 10);

        RCLCPP_INFO(this->get_logger(), "等待图像……");
    }

private:
    void onImage(const sensor_msgs::msg::CompressedImage::SharedPtr msg)
    {
        // 1. 解码
        cv::Mat img = cv::imdecode(msg->data, cv::IMREAD_COLOR);
        if (img.empty()) {
            RCLCPP_WARN_THROTTLE(this->get_logger(), *this->get_clock(), 5000,
                                 "解码失败");
            return;
        }

        // 2. 灰度 + 平均亮度
        cv::Mat gray;
        cv::cvtColor(img, gray, cv::COLOR_BGR2GRAY);
        double brightness = cv::mean(gray)[0];    // 0~255

        // 3. 发布
        std_msgs::msg::Float32 out;
        out.data = static_cast<float>(brightness);
        pub_->publish(out);

        // 4. 限频日志 + 显示
        RCLCPP_INFO_THROTTLE(this->get_logger(), *this->get_clock(), 2000,
                             "亮度: %.1f", brightness);

        static int count = 0;
        if (++count % 3 == 0) {
            cv::putText(img, "brightness=" + std::to_string((int)brightness),
                        cv::Point(10, 30), cv::FONT_HERSHEY_SIMPLEX,
                        0.8, cv::Scalar(0, 255, 0), 2);
            cv::imshow("brightness", img);
            cv::waitKey(1);
        }
    }

    rclcpp::Subscription<sensor_msgs::msg::CompressedImage>::SharedPtr sub_;
    rclcpp::Publisher<std_msgs::msg::Float32>::SharedPtr pub_;
};

int main(int argc, char** argv)
{
    rclcpp::init(argc, argv);
    rclcpp::spin(std::make_shared<BrightnessNode>());
    rclcpp::shutdown();
    return 0;
}
```

### 3.2 加进 CMakeLists.txt

```cmake
find_package(OpenCV REQUIRED)     # 顶部和其他 find_package 放一起

add_executable(brightness_node src/brightness_node.cpp)
ament_target_dependencies(brightness_node rclcpp sensor_msgs std_msgs)
target_link_libraries(brightness_node ${OpenCV_LIBS})

install(TARGETS talker listener brightness_node
  DESTINATION lib/${PROJECT_NAME})
```

### 3.3 编译运行

```bash
cd ~/practice_ws
colcon build --packages-select my_nodes
source install/setup.bash

ros2 run my_nodes brightness_node
```

## 5. 自己动手

### 挑战 1：写一个假摄像头（发布假图像）

在 `talker.cpp` 的基础上改：定时器里生成一张 `Mat`（画个红圆），
`imencode` 成 JPEG，发到 `/camera/image_raw/compressed`。
然后 `brightness_node` 就有数据了 —— **这招能让你在没有硬件/游戏时独立测试。**

```cpp
cv::Mat img(480, 640, CV_8UC3, cv::Scalar(30, 30, 30));
cv::circle(img, cv::Point(320, 240), 120, cv::Scalar(0, 0, 255), -1);
std::vector<uchar> buf;
cv::imencode(".jpg", img, buf);

sensor_msgs::msg::CompressedImage msg;
msg.format = "jpeg";
msg.data = buf;
pub_->publish(msg);
```

### 挑战 2：加参数

把"显示开关"做成参数：`ros2 run my_nodes brightness_node --ros-args -p show:=true`。

### 挑战 3：发布灰度图

把灰度图编码后发到 `/debug_gray`，在别的终端 `ros2 topic echo --no-arr` 查看。

### 挑战 4：两个节点接力

假摄像头 → brightness_node → 再写个 `alert_node`：亮度低于阈值就打印"太暗了！"
（三个节点一条链，体会"每个节点只做一件事"）
